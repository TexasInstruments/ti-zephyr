194__syscall int led_set_brightness(const struct device *dev, uint32_t led,
195 uint8_t value);
196
197static inline int z_impl_led_set_brightness(const struct device *dev,
198 uint32_t led,
199 uint8_t value)
200{
201 const struct led_driver_api *api =
202 (const struct led_driver_api *)dev->api;
203
204 if (api->set_brightness == NULL) {
205 if (api->on == NULL || api->off == NULL) {
206 return -ENOSYS;
207 }
208 }
209
210 if (value > LED_BRIGHTNESS_MAX) {
211 return -EINVAL;
212 }
213
214 if (api->set_brightness == NULL) {
215 if (value) {
216 return api->on(dev, led);
217 } else {
218 return api->off(dev, led);
219 }
220 }
221
222 return api->set_brightness(dev, led, value);
223}

322__syscall int led_on(const struct device *dev, uint32_t led);
323
324static inline int z_impl_led_on(const struct device *dev, uint32_t led)
325{
326 const struct led_driver_api *api =
327 (const struct led_driver_api *)dev->api;
328
329 if (api->set_brightness == NULL && api->on == NULL) {
330 return -ENOSYS;
331 }
332
333 if (api->on == NULL) {
334 return api->set_brightness(dev, led, LED_BRIGHTNESS_MAX);
335 }
336
337 return api->on(dev, led);
338}
