66static ALWAYS_INLINE unsigned int z_sparc_set_pil_inline(unsigned int newpil)
67{
68 register uint32_t oldpil __asm__ ("o0") = newpil;
69
70 __asm__ volatile (
71 "ta %1\nnop\n" :
72 "=r" (oldpil) :
73 "i" (SPARC_SW_TRAP_SET_PIL), "r" (oldpil) :
74 "memory"
75 );
76 return oldpil;
77}
