9/*
10 * The whole TI CC35XX pin configuration information is encoded in a 32-bit
11 * bitfield organized as follows:
12 *
13 * - 31..22: Reserved.
14 * - 21..16: Pin.
15 * - 15..11: Reserved.
16 * - 11..10: Pin direction.
17 * - 9: Pull-down flag.
18 * - 8: Pull-up flag.
19 * - 7..5: Drive strength.
20 * - 4: Enable open-drain flag.
21 * - 3..0: Configuration mode.
22 */
23

29#define TI_CC35XX_PIN_MSK 0x3f
30#define TI_CC35XX_PIN_POS 16
31#define TI_CC35XX_OUT_OVR_MSK 0x03
32#define TI_CC35XX_OUT_OVR_POS 12
33#define TI_CC35XX_DIR_MSK 0x03
34#define TI_CC35XX_DIR_POS 10
35#define TI_CC35XX_PU_PD_MSK 0x03
36#define TI_CC35XX_PU_PD_POS 8
37#define TI_CC35XX_MUX_MSK 0x0f
38#define TI_CC35XX_MUX_POS 0

42#define TI_CC35XX_PINCTRL_DIR_BI (TI_CC35XX_PINCTRL_DIR_IN | TI_CC35XX_PINCTRL_DIR_OUT)
43#define TI_CC35XX_PINCTRL_DIR_IN (2 << TI_CC35XX_DIR_POS)
44#define TI_CC35XX_PINCTRL_DIR_OUT (1 << TI_CC35XX_DIR_POS)
45#define TI_CC35XX_PINCTRL_PULL_UP (1 << TI_CC35XX_PU_PD_POS)
46#define TI_CC35XX_PINCTRL_PULL_DOWN (2 << TI_CC35XX_PU_PD_POS)
47#define TI_CC35XX_PINCTRL_PULL_DISABLE (3 << TI_CC35XX_PU_PD_POS)
48
49#define TI_CC35XX_PINCTRL_PIN(PCFG) (((PCFG) >> TI_CC35XX_PIN_POS) & TI_CC35XX_PIN_MSK)
50#define TI_CC35XX_PINCTRL_MUX(PCFG) (((PCFG) >> TI_CC35XX_MUX_POS) & TI_CC35XX_MUX_MSK)
51
52#define TI_CC35XX_PINCTRL_OVR_SET_VAL(en, val) (((en << 1) | val) << TI_CC35XX_OUT_OVR_POS)
53#define TI_CC35XX_PINCTRL_OVR_VAL(PCTL) (((PCTL) >> TI_CC35XX_OUT_OVR_POS) & TI_CC35XX_OUT_OVR_MSK)
54
55#define TI_CC35XX_PINCTRL_OVR_NO_CHANGE TI_CC35XX_PINCTRL_OVR_SET_VAL(0, 0)
56#define TI_CC35XX_PINCTRL_OVR_DISABLE TI_CC35XX_PINCTRL_OVR_SET_VAL(0, 1)
57#define TI_CC35XX_PINCTRL_OVR_ENABLE_LOW TI_CC35XX_PINCTRL_OVR_SET_VAL(1, 0)
58#define TI_CC35XX_PINCTRL_OVR_ENABLE_HIGH TI_CC35XX_PINCTRL_OVR_SET_VAL(1, 1)

66#define TI_CC35XX_PINMUX(pin, mux) \
67 ((((pin) & TI_CC35XX_PIN_MSK) << TI_CC35XX_PIN_POS) | \
68 (((mux) & TI_CC35XX_MUX_MSK) << TI_CC35XX_MUX_POS))

70#define TI_CC35XX_ANALOG_MUX 0x0d
71#define TI_CC35XX_ANALOG(pin) \
