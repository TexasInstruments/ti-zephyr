890static inline int display_register_event_cb(const struct device *dev,
891 display_event_cb_t cb, void *user_data,
892 uint32_t event_mask, bool in_isr,
893 uint32_t *out_reg_handle)
894{
895 __ASSERT(cb != NULL, "Registration failed: callback function pointer is NULL");
896
897 const struct display_driver_api *api = DEVICE_API_GET(display, dev);
898 struct display_capabilities caps;
899
900 if (api->register_event_cb == NULL) {
901 return -ENOSYS;
902 }
903
904 display_get_capabilities(dev, &caps);
905 if (!caps.supported_events) {
906 return -ENOSYS;
907 }
908 if (event_mask == 0 || (~caps.supported_events & event_mask)) {
909 return -EINVAL;
910 }
911
912 return api->register_event_cb(dev, cb, user_data, event_mask, in_isr, out_reg_handle);
913}

926static inline int display_unregister_event_cb(const struct device *dev, uint32_t reg_handle)
927{
928 const struct display_driver_api *api = DEVICE_API_GET(display, dev);
929
930 if (api->unregister_event_cb == NULL || api->register_event_cb == NULL) {
931 return -ENOSYS;
932 }
933
934 if (reg_handle == 0U) {
935 return -EINVAL;
936 }
937
938 return api->unregister_event_cb(dev, reg_handle);
939}
