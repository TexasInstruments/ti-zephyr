9/*
10 * Packed mux-state trailing cell for the AM13E2 input X-BAR: the GPIO
11 * number to route, plus whether that GPIO's RAW_INPUTSELECT_Pn bit should
12 * select the raw pad signal (1) or the signal after the GPIO peripheral
13 * module processes it (0, TRM reset default). The bit lives per-GPIO, not
14 * per-channel - routing two channels to the same GPIO with conflicting
15 * raw settings is a devicetree authoring error, not something the driver
16 * arbitrates.
17 */
18#define AM13E2_XBAR_IN_PIN_MASK 0xFFU
19#define AM13E2_XBAR_IN_RAW_SHIFT 8U
20#define AM13E2_XBAR_IN_RAW_MASK 0x1U
21
22#define AM13E2_XBAR_IN_STATE(pin, raw) \
23 (((pin) & AM13E2_XBAR_IN_PIN_MASK) | \
24 (((raw) & AM13E2_XBAR_IN_RAW_MASK) << AM13E2_XBAR_IN_RAW_SHIFT))
25
26#define AM13E2_XBAR_IN_STATE_PIN(state) ((state) & AM13E2_XBAR_IN_PIN_MASK)
27#define AM13E2_XBAR_IN_STATE_RAW(state) \
28 (((state) >> AM13E2_XBAR_IN_RAW_SHIFT) & AM13E2_XBAR_IN_RAW_MASK)
29
30#endif /* ZEPHYR_INCLUDE_DT_BINDINGS_MUX_TI_AM13E2_XBAR_IN_H_ */
