144__pinned_func
145static inline bool arch_is_user_context(void)
146{
147 int cs;
148
149 /* On x86, read the CS register (which cannot be manually set) */
150 __asm__ volatile ("mov %%cs, %[cs_val]" : [cs_val] "=r" (cs));
151
152 return cs == USER_CODE_SEG;
153}
