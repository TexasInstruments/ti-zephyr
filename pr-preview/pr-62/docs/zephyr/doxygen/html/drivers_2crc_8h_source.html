279static inline int crc_verify(struct crc_ctx *ctx, crc_result_t expected)
280{
281 if (ctx == NULL) {
282 return -EINVAL;
283 }
284
285 if (ctx->state == CRC_STATE_IN_PROGRESS) {
286 return -EBUSY;
287 }
288
289 if (expected != ctx->result) {
290 return -EPERM;
291 }
292
293 return 0;
294}
