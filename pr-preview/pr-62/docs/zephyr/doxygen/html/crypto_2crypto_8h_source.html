87/* Following are the public API a user app may call.
88 * The first two relate to crypto "session" setup / teardown. Further we
89 * have four cipher mode specific (CTR, CCM, CBC ...) calls to perform the
90 * actual crypto operation in the context of a session. Also we have an
91 * API to provide the callback for async operations.

105static inline int crypto_query_hwcaps(const struct device *dev)
106{
107 struct crypto_driver_api *api;
108 int tmp;
109
110 api = (struct crypto_driver_api *)dev->api;
111
112 tmp = api->query_hw_caps(dev);
113
114 __ASSERT((tmp & (CAP_OPAQUE_KEY_HNDL | CAP_RAW_KEY)) != 0,
115 "Driver should support at least one key type: RAW/Opaque");
116
117 __ASSERT((tmp & (CAP_INPLACE_OPS | CAP_SEPARATE_IO_BUFS)) != 0,
118 "Driver should support at least one IO buf type: Inplace/separate");
119
120 __ASSERT((tmp & (CAP_SYNC_OPS | CAP_ASYNC_OPS)) != 0,
121 "Driver should support at least one op-type: sync/async");
122 return tmp;
123}

155static inline int cipher_begin_session(const struct device *dev, struct cipher_ctx *ctx,
156 enum cipher_algo algo, enum cipher_mode mode,
157 enum cipher_op optype)
158{
159 struct crypto_driver_api *api;
160 uint32_t flags;
161
162 api = (struct crypto_driver_api *)dev->api;
163 ctx->device = dev;
164 ctx->ops.cipher_mode = mode;
165
166 flags = (ctx->flags & (CAP_OPAQUE_KEY_HNDL | CAP_RAW_KEY));
167 __ASSERT(flags != 0U, "Keytype missing: RAW Key or OPAQUE handle");
168 __ASSERT(flags != (CAP_OPAQUE_KEY_HNDL | CAP_RAW_KEY), "conflicting options for keytype");
169
170 flags = (ctx->flags & (CAP_INPLACE_OPS | CAP_SEPARATE_IO_BUFS));
171 __ASSERT(flags != 0U, "IO buffer type missing");
172 __ASSERT(flags != (CAP_INPLACE_OPS | CAP_SEPARATE_IO_BUFS),
173 "conflicting options for IO buffer type");
174
175 flags = (ctx->flags & (CAP_SYNC_OPS | CAP_ASYNC_OPS));
176 __ASSERT(flags != 0U, "sync/async type missing");
177 __ASSERT(flags != (CAP_SYNC_OPS | CAP_ASYNC_OPS), "conflicting options for sync/async");
178
179 return api->cipher_begin_session(dev, ctx, algo, mode, optype);
180}
