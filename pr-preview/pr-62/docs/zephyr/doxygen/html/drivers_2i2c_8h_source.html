642static inline void i2c_xfer_stats(const struct device *dev, struct i2c_msg *msgs,
643 uint8_t num_msgs)
644{
645 struct i2c_device_state *state =
646 CONTAINER_OF(dev->state, struct i2c_device_state, devstate);
647 uint32_t bytes_read = 0U;
648 uint32_t bytes_written = 0U;
649
650 STATS_INC(state->stats, transfer_call_count);
651 STATS_INCN(state->stats, message_count, num_msgs);
652 for (uint8_t i = 0U; i < num_msgs; i++) {
653 if (msgs[i].flags & I2C_MSG_READ) {
654 bytes_read += msgs[i].len;
655 } else {
656 bytes_written += msgs[i].len;
657 }
658 }
659 STATS_INCN(state->stats, bytes_read, bytes_read);
660 STATS_INCN(state->stats, bytes_written, bytes_written);
661}

888__syscall int i2c_transfer(const struct device *dev,
889 struct i2c_msg *msgs, uint8_t num_msgs,
890 uint16_t addr);
891
892static inline int z_impl_i2c_transfer(const struct device *dev,
893 struct i2c_msg *msgs, uint8_t num_msgs,
894 uint16_t addr)
895{
896 const struct i2c_driver_api *api =
897 (const struct i2c_driver_api *)dev->api;
898
899 if (!num_msgs) {
900 return 0;
901 }
902
903 if (!IS_ENABLED(CONFIG_I2C_ALLOW_NO_STOP_TRANSACTIONS)) {
904 msgs[num_msgs - 1].flags |= I2C_MSG_STOP;
905 }
906
907 int res = api->transfer(dev, msgs, num_msgs, addr);
908
909 i2c_xfer_stats(dev, msgs, num_msgs);
910
911 if (IS_ENABLED(CONFIG_I2C_DUMP_MESSAGES)) {
912 i2c_dump_msgs_rw(dev, msgs, num_msgs, addr, true);
913 }
914
915 return res;
916}

1015static inline int i2c_write_read_cb(const struct device *dev, struct i2c_msg *msgs,
1016 uint8_t num_msgs, uint16_t addr, const void *write_buf,
1017 size_t num_write, void *read_buf, size_t num_read,
1018 i2c_callback_t cb, void *userdata)
1019{
1020 if ((msgs == NULL) || (num_msgs != 2)) {
1021 return -EINVAL;
1022 }
1023
1024 msgs[0].buf = (uint8_t *)write_buf;
1025 msgs[0].len = num_write;
1026 msgs[0].flags = I2C_MSG_WRITE;
1027
1028 msgs[1].buf = (uint8_t *)read_buf;
1029 msgs[1].len = num_read;
1030 msgs[1].flags = I2C_MSG_RESTART | I2C_MSG_READ | I2C_MSG_STOP;
1031
1032 return i2c_transfer_cb(dev, msgs, num_msgs, addr, cb, userdata);
1033}
